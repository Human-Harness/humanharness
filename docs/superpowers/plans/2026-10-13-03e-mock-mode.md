**Goal:** `humanharness dev --mock` opens the app with a fake engine that replays hand-editable recordings (`<harness>/mock/<action>.json`), spending no tokens. `run --record` writes those recordings, `check` validates them, and the demo's recordings drive an e2e test.

**Architecture:** The recording schema and the replay logic live in `packages/fake-agent`. `fake-claude` gets a mock mode, driven by `HH_FAKE_MOCK_DIR` and `HH_FAKE_MOCK_STATE`: it picks the recording by the first message, walks the turn tree by the person's reply, and prints Claude stream-json, calls the `ui` MCP tools and writes files. The shell needs no new engine path; it reuses the existing dev-only test hooks (`HH_TEST_*`). Two additions there: `HH_TEST_MOCK` (offer only the faked engines, show a badge) and `HH_TEST_SUGGESTED_FOLDER`. The CLI wires it all up: `dev --mock` builds the env, `run --record` taps the TurnController's events, and `check` gains the `mock-*` rules.

// packages/fake-agent/src/mock.ts
import { readFileSync, readdirSync } from "node:fs";
import path from "node:path";
import { z } from "zod";

/** A path inside the work folder: relative, with no `..` part. */
const folderPath = z
  .string()
  .min(1)
  .refine((p) => !path.isAbsolute(p) && !p.split(/[\\/]/).includes(".."), "must be a relative path inside the folder");

const blockSchema = z.strictObject({ tool: z.string().min(1), input: z.record(z.string(), z.unknown()) });
const writeSchema = z.strictObject({ path: folderPath, content: z.string().nullable() });
const usageSchema = z.strictObject({
  input: z.number().int().nonnegative(),
  output: z.number().int().nonnegative(),
  cacheRead: z.number().int().nonnegative().default(0),
  cacheWrite: z.number().int().nonnegative().default(0),
});
/** The `save_note` arguments; the app validates them as it does for a real engine. */
const noteSchema = z.looseObject({ title: z.string().min(1) });

export const mockTurnSchema = z.strictObject({
  id: z.string().min(1),
  text: z.string().default(""),
  blocks: z.array(blockSchema).default([]),
  note: noteSchema.optional(),
  writes: z.array(writeSchema).default([]),
  usage: usageSchema.optional(),
  delayMs: z.number().int().min(0).max(60_000).optional(),
  /** Reply → next turn id: a choice's label, or `*` for anything else. */
  next: z.record(z.string(), z.string()).optional(),
});

/** One action's recorded conversation (spec 3e §2). */
export const recordingSchema = z.strictObject({
  version: z.literal(1),
  action: z.string().min(1),
  /** Plain text that the task's first message (its brief) contains. */
  match: z.string().min(1),
  recordedWith: z.string().optional(),
  fallback: z.strictObject({ text: z.string().min(1) }).optional(),
  turns: z.array(mockTurnSchema).min(1),
});

export type Recording = z.infer<typeof recordingSchema>;
export type MockTurn = z.infer<typeof mockTurnSchema>;
export type MockBlock = z.infer<typeof blockSchema>;
export type MockWrite = z.infer<typeof writeSchema>;
export type MockUsage = z.infer<typeof usageSchema>;
export type MockNote = z.infer<typeof noteSchema>;

/** Every `*.json` in `dir` that parses as a recording; the others are named in `problems`. */
export function loadRecordings(dir: string): { recordings: Recording[]; problems: string[] } {
  let names: string[];
  try {
    names = readdirSync(dir).filter((n) => n.endsWith(".json")).sort();
  } catch {
    return { recordings: [], problems: [] };
  }
  const recordings: Recording[] = [];
  const problems: string[] = [];
  for (const name of names) {
    try {
      const parsed = recordingSchema.safeParse(JSON.parse(readFileSync(path.join(dir, name), "utf8")));
      if (parsed.success) recordings.push(parsed.data);
      else problems.push(`${name}: ${parsed.error.issues.map((i) => `${i.path.join(".") || "(root)"} ${i.message}`).join("; ")}`);
    } catch (err) {
      problems.push(`${name}: ${(err as Error).message}`);
    }
  }
  return { recordings, problems };
}
```

What the app sends (from `packages/shell/src/main/task-prompts.ts` and `task-manager.ts`):
- **First message of a task:** the action's filled brief, which starts with a line like `# Topic: the manifest`. A resumed task's first message also contains the brief. A free question starts with `The person asked a question`.
- **A choice click:** the option's `followUp`, or its `label` when it has none. An option with `next: true` or an `action` closes the task (closing turn) and starts another task in a new session.
- **The closing turn:** `CLOSE_PROMPT`, which starts with `We are closing this topic.`
- **A checkpoint:** `CHECKPOINT_PROMPT`, which starts with `Checkpoint:`.

// packages/fake-agent/src/mock-replay.ts
import type { MockBlock, MockNote, MockTurn, MockUsage, MockWrite, Recording } from "./mock";

/** Played when a reply isn't recorded and the recording has no `fallback` (spec 3e §2). */
export const DEFAULT_FALLBACK = "Respuesta no grabada en esta prueba · Not recorded in this test.";
const FALLBACK_USAGE: MockUsage = { input: 400, output: 20, cacheRead: 0, cacheWrite: 0 };
const DEFAULT_DELAY = 600;

/** Where one engine session is: its recording's action and the last turn played (null: none matched). */
export interface SessionPosition {
  action: string | null;
  turn: string | null;
}

/** What the fake prints and does for one engine turn. */
export interface Play {
  text: string;
  blocks: MockBlock[];
  note?: MockNote;
  writes: MockWrite[];
  usage: MockUsage;
  delayMs: number;
}

/**
 * The app's fixed prompts start with these (CLOSE_PROMPT and CHECKPOINT_PROMPT in the shell's
 * task-prompts.ts; a shell test pins them).
 */
export function promptKind(prompt: string): "close" | "checkpoint" | "reply" {
  if (prompt.startsWith("We are closing this topic.")) return "close";
  if (prompt.startsWith("Checkpoint:")) return "checkpoint";
  return "reply";
}

const defaultNote = (action: string): MockNote => ({ title: action, summary: ["Recorded test run"], decisions: [], files: [], open: [] });
const firstTurn = (r: Recording) => r.turns.find((t) => t.id !== "close");
const closeNote = (r: Recording) => r.turns.find((t) => t.id === "close")?.note ?? defaultNote(r.action);
const choicesOf = (turn: MockTurn | undefined) => turn?.blocks.filter((b) => b.tool === "show_choices") ?? [];

function playTurn(turn: MockTurn): Play {
  return {
    text: turn.text,
    blocks: turn.blocks,
    ...(turn.note ? { note: turn.note } : {}),
    writes: turn.writes,
    usage: turn.usage ?? FALLBACK_USAGE,
    delayMs: turn.delayMs ?? DEFAULT_DELAY,
  };
}

const quiet = (text: string, blocks: MockBlock[] = [], note?: MockNote): Play => ({
  text,
  blocks,
  ...(note ? { note } : {}),
  writes: [],
  usage: FALLBACK_USAGE,
  delayMs: DEFAULT_DELAY / 2,
});

/** The label a reply stands for: the option whose followUp (or, without one, label) is the reply. */
function replyKey(turn: MockTurn, prompt: string): string | undefined {
  const reply = prompt.trim();
  for (const block of choicesOf(turn)) {
    const options = Array.isArray(block.input.options) ? (block.input.options as { label?: unknown; followUp?: unknown }[]) : [];
    const hit = options.find((o) => typeof o.label === "string" && (typeof o.followUp === "string" ? o.followUp.trim() : o.label.trim()) === reply);
    if (hit) return hit.label as string;
  }
  return undefined;
}

/**
 * One engine turn of mock mode. `newSession`: the run opened a session (`--session-id`), so the prompt is a
 * task's first message and picks the recording; otherwise `position` says where the session is.
 */
export function replay(recordings: Recording[], position: SessionPosition | undefined, prompt: string, newSession: boolean): { play: Play; session: SessionPosition } {
  if (newSession || !position) {
    // The longest match wins, so "# Topic: engines" never shadows "# Topic: engines and more".
    const rec = [...recordings].sort((a, b) => b.match.length - a.match.length).find((r) => prompt.includes(r.match));
    const turn = rec && firstTurn(rec);
    if (!rec || !turn) return { play: quiet(DEFAULT_FALLBACK), session: { action: rec?.action ?? null, turn: null } };
    return { play: playTurn(turn), session: { action: rec.action, turn: turn.id } };
  }
  const rec = recordings.find((r) => r.action === position.action);
  const kind = promptKind(prompt);
  if (kind === "close") {
    const close = rec?.turns.find((t) => t.id === "close");
    if (close) return { play: playTurn({ ...close, blocks: [] }), session: position };
    return { play: quiet("", [], defaultNote(position.action ?? "question")), session: position };
  }
  if (kind === "checkpoint") return { play: quiet("", [], rec ? closeNote(rec) : defaultNote(position.action ?? "question")), session: position };
  const current = rec?.turns.find((t) => t.id === position.turn);
  const key = current ? replyKey(current, prompt) : undefined;
  const target = current?.next && ((key !== undefined && Object.hasOwn(current.next, key) ? current.next[key] : undefined) ?? current.next["*"]);
  const turn = target ? rec?.turns.find((t) => t.id === target) : undefined;
  if (turn) return { play: playTurn(turn), session: { ...position, turn: turn.id } };
  return { play: quiet(rec?.fallback?.text ?? DEFAULT_FALLBACK, choicesOf(current)), session: position };
}
```

// packages/fake-agent/src/mock-claude.ts
// fake-claude's mock mode (spec 3e): replays <harness>/mock/*.json instead of a test script.
import { mkdirSync, readFileSync, rmSync, writeFileSync } from "node:fs";
import path from "node:path";
import { callMcpTool } from "./mcp-call";
import { loadRecordings } from "./mock";
import { type SessionPosition, replay } from "./mock-replay";

type McpServers = Record<string, { command: string; args?: string[]; env?: Record<string, string> }>;

function readState(file: string): Record<string, SessionPosition> {
  try {
    const parsed = JSON.parse(readFileSync(file, "utf8")) as { sessions?: Record<string, SessionPosition> };
    return parsed.sessions ?? {};
  } catch {
    return {};
  }
}

const flag = (argv: string[], name: string) => (argv.includes(name) ? argv[argv.indexOf(name) + 1] : undefined);
const sleep = (ms: number) => new Promise((r) => setTimeout(r, ms));
const print = (line: unknown) => process.stdout.write(JSON.stringify(line) + "\n");

/** One engine turn in mock mode: prints Claude's stream-json, calls the ui tools and writes the files. */
export async function runMockTurn(argv: string[], prompt: string, mcpServers: McpServers, env: { dir: string; state: string }): Promise<void> {
  const resumed = flag(argv, "--resume");
  const sessionId = resumed ?? flag(argv, "--session-id") ?? "mock-session";
  const { recordings, problems } = loadRecordings(env.dir);
  for (const p of problems) process.stderr.write(`mock: skipped ${p}\n`);
  const sessions = readState(env.state);
  const { play, session } = replay(recordings, sessions[sessionId], prompt, resumed === undefined);
  sessions[sessionId] = session;
  mkdirSync(path.dirname(env.state), { recursive: true });
  writeFileSync(env.state, JSON.stringify({ sessions }));

  const model = flag(argv, "--model");
  print({ type: "system", subtype: "init", session_id: sessionId, ...(model ? { model } : {}) });
  for (const w of play.writes) {
    const target = path.resolve(w.path);
    if (w.content === null) rmSync(target, { force: true });
    else {
      mkdirSync(path.dirname(target), { recursive: true });
      writeFileSync(target, w.content);
    }
  }
  if (play.text) {
    const event = (e: unknown) => print({ type: "stream_event", parent_tool_use_id: null, session_id: sessionId, event: e });
    // Streamed in about eight chunks over delayMs, so the window shows it arriving.
    const words = play.text.split(/(?<=\s)/);
    const size = Math.max(1, Math.ceil(words.length / 8));
    event({ type: "content_block_start", index: 0, content_block: { type: "text", text: "" } });
    for (let i = 0; i < words.length; i += size) {
      event({ type: "content_block_delta", index: 0, delta: { type: "text_delta", text: words.slice(i, i + size).join("") } });
      await sleep(play.delayMs / Math.ceil(words.length / size));
    }
    event({ type: "content_block_stop", index: 0 });
  }
  const ui = mcpServers.ui;
  const call = async (tool: string, args: Record<string, unknown>) => {
    if (!ui) throw new Error("fake-claude mock: the app configured no ui MCP server");
    await callMcpTool({ command: ui.command, args: ui.args ?? [], env: ui.env ?? {} }, tool, args);
  };
  for (const b of play.blocks) await call(b.tool, b.input);
  if (play.note) await call("save_note", play.note);
  const u = play.usage;
  print({
    type: "result",
    subtype: "success",
    is_error: false,
    result: play.text,
    session_id: sessionId,
    usage: { input_tokens: u.input, output_tokens: u.output, cache_read_input_tokens: u.cacheRead, cache_creation_input_tokens: u.cacheWrite },
  });
}

export async function mockEnv(o: { harnessId: string; harnessDir: string; suggested: string; keep: boolean; out: { write(s: string): unknown } }): Promise<Record<string, string>> {

// packages/cli/src/record.ts
import { promises as fs } from "node:fs";
import path from "node:path";
import type { MockNote, MockTurn, MockUsage, MockWrite, Recording } from "@humanharness/fake-agent";
import type { Event } from "@humanharness/runtime";

/** Larger files, and files that aren't text, are left out of a recording (spec 3e §3). */
export const MAX_RECORDED_FILE = 64 * 1024;
const SKIPPED_DIRS = new Set([".harness", ".git", "node_modules"]);

export interface FolderSnapshot {
  /** Relative path (with /) → content. */
  files: Map<string, string>;
  /** Relative paths left out: binary or larger than MAX_RECORDED_FILE. */
  skipped: Set<string>;
}

/** The folder's small text files, `.harness/` left out. */
export async function snapshotFolder(folder: string): Promise<FolderSnapshot> {
  const snap: FolderSnapshot = { files: new Map(), skipped: new Set() };
  const walk = async (dir: string): Promise<void> => {
    const entries = await fs.readdir(dir, { withFileTypes: true }).catch(() => []);
    for (const e of entries) {
      const abs = path.join(dir, e.name);
      const rel = path.relative(folder, abs).split(path.sep).join("/");
      if (e.isDirectory()) {
        if (!SKIPPED_DIRS.has(e.name)) await walk(abs);
      } else if (e.isFile()) {
        const stat = await fs.stat(abs);
        if (stat.size > MAX_RECORDED_FILE) {
          snap.skipped.add(rel);
          continue;
        }
        const buf = await fs.readFile(abs);
        if (buf.includes(0)) snap.skipped.add(rel);
        else snap.files.set(rel, buf.toString("utf8"));
      }
    }
  };
  await walk(folder);
  return snap;
}

function diff(before: FolderSnapshot, after: FolderSnapshot): MockWrite[] {
  const writes: MockWrite[] = [];
  for (const [p, content] of after.files) if (before.files.get(p) !== content) writes.push({ path: p, content });
  for (const p of before.files.keys()) if (!after.files.has(p) && !after.skipped.has(p)) writes.push({ path: p, content: null });
  return writes.sort((a, b) => a.path.localeCompare(b.path));
}

export interface RecorderOptions {
  folder: string;
  /** The MCP tool a block is shown with (`blockToolName` over the harness's blocks). */
  toolFor: (block: string) => string;
  /** The text a task's first message contains for this action (its brief's first line). */
  matchFor: (action: string) => string;
  recordedWith: string;
}

/** Builds recordings from one `run`: what the engine answered, what the person replied, what changed. */
export class Recorder {
  private readonly byAction = new Map<string, Recording>();
  private text = "";
  private blocks: MockTurn["blocks"] = [];
  private usage: MockUsage | undefined;
  private lastNote: MockNote | undefined;
  private replyKey: string | undefined;
  private lastTurn: { action: string; turn: MockTurn } | undefined;
  /** A closing turn that ended before its note was written. */
  private pendingClose: MockTurn | undefined;
  private snapshot: FolderSnapshot = { files: new Map(), skipped: new Set() };
  /** Every path skipped at least once, for the run's output. */
  readonly skipped = new Set<string>();

  constructor(private readonly o: RecorderOptions) {}

  /** The folder as it is before the first turn. */
  async begin(): Promise<void> {
    this.snapshot = await snapshotFolder(this.o.folder);
  }

  event(e: Event): void {
    if (e.type === "text") this.text += e.delta;
    else if (e.type === "paragraph") this.text += "\n\n";
    else if (e.type === "usage") this.usage = { input: e.inputTokens, output: e.outputTokens, cacheRead: e.cacheReadTokens, cacheWrite: e.cacheWriteTokens };
    else if (e.type === "block" && e.name !== "save_note" && typeof e.data === "object" && e.data !== null) {
      this.blocks.push({ tool: this.o.toolFor(e.name), input: e.data as Record<string, unknown> });
    }
  }

  /** What the person answered the last turn with: a choice's label, or `*` for typed text. */
  reply(key: string): void {
    this.replyKey = key;
  }

  /** A note TaskManager wrote. It may come before or after its closing turn's `turnEnded`. */
  note(note: MockNote): void {
    if (this.pendingClose) {
      this.pendingClose.note = note;
      this.pendingClose = undefined;
    } else this.lastNote = note;
  }

  /** One engine turn ended (`recordUsage`): it becomes a recorded turn unless it was a checkpoint. */
  async turnEnded(meta: { action?: string; phase?: "run" | "close" | "checkpoint" }): Promise<void> {
    const after = await snapshotFolder(this.o.folder);
    for (const p of after.skipped) this.skipped.add(p);
    const writes = diff(this.snapshot, after);
    this.snapshot = after;
    const action = meta.action;
    const taken = { text: this.text, blocks: this.blocks, usage: this.usage };
    this.text = "";
    this.blocks = [];
    this.usage = undefined;
    if (!action || meta.phase === "checkpoint") return;
    let rec = this.byAction.get(action);
    if (!rec) {
      rec = { version: 1, action, match: this.o.matchFor(action), recordedWith: this.o.recordedWith, turns: [] };
      this.byAction.set(action, rec);
    }
    const id = meta.phase === "close" ? "close" : `t${rec.turns.filter((t) => t.id !== "close").length + 1}`;
    const turn: MockTurn = {
      id,
      text: taken.text,
      blocks: taken.blocks,
      writes,
      ...(taken.usage ? { usage: taken.usage } : {}),
      ...(meta.phase === "close" && this.lastNote ? { note: this.lastNote } : {}),
    };
    rec.turns.push(turn);
    if (meta.phase !== "close" && this.lastTurn?.action === action && this.replyKey !== undefined) {
      this.lastTurn.turn.next = { ...this.lastTurn.turn.next, [this.replyKey]: id };
    }
    this.replyKey = undefined;
    if (meta.phase === "close") {
      if (!turn.note) this.pendingClose = turn;
      this.lastNote = undefined;
      this.lastTurn = undefined;
    } else this.lastTurn = { action, turn };
  }

  recordings(): Recording[] {
    return [...this.byAction.values()];
  }
}
